`timescale 1ns / 1ps

module vending_machine_tb;

    reg clk, rst, coin5, coin10;
    wire dispense, change5;

    vending_machine dut (
        .clk(clk),
        .rst(rst),
        .c5(coin5),
        .c10(coin10),
        .out(dispense),
        .change5(change5)
    );

    initial begin
        clk = 0;
        forever #5 clk = ~clk;
    end

    task insert_coin;
        input c5;
        input c10;
        begin
            @(negedge clk);
            coin5 = c5;
            coin10 = c10;

            @(posedge clk);
            #1;

            $display(
                "Time=%0t coin5=%b coin10=%b dispense=%b change5=%b",
                $time, coin5, coin10, dispense, change5
            );

            @(negedge clk);
            coin5 = 0;
            coin10 = 0;
        end
    endtask

    initial begin
        rst = 1;
        coin5 = 0;
        coin10 = 0;

        repeat (2) @(negedge clk);
        rst = 0;

        // Transaction 1: ₹5 + ₹10 = ₹15
        insert_coin(1, 0);
        insert_coin(0, 1);

        // Transaction 2: ₹10 + ₹5 = ₹15
        insert_coin(0, 1);
        insert_coin(1, 0);

        // Transaction 3: ₹10 + ₹10 = ₹20
        // Dispense product and return ₹5
        insert_coin(0, 1);
        insert_coin(0, 1);

        $display("Vending machine test completed.");
        $finish;
    end

endmodule
